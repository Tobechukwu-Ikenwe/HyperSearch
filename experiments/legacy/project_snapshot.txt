#ifndef SIMILARITY_HPP
#define SIMILARITY_HPP

#include "embedding.hpp"
#include <stdexcept>
#include <cmath>

// dot product how similar 2 vectors are

float dot_product(const Embedding& A, const Embedding& B){
	//check if the vector size is the same
	if (A.size() != B.size()){
	throw std::invalid_argument("Embeddings must have the same size");
	}
	
	//then loop through size multiplying each corresponding value in both matrices
	//e.g. a[i] * b[i] then add to sum
	//
	float result = 0.0f;
	for (size_t i = 0; i < A.size(); i++){
		result += (A[i] * B[i]);
	}
	return result;
}


// cosine similarity - true similarity. dividing the dot product by the product of the vector magnitudes

float magnitude(const Embedding& A){
	float sum = 0.0f;
	for (size_t i = 0; i < A.size(); i++){
		sum += A[i] * A[i];
	}
	
	return std::sqrt(sum);
}

float cosine_similarity(const Embedding& A, const Embedding& B){
	float magA = magnitude(A);
	float magB = magnitude(B);

	if (magA == 0.0f || magB == 0.0f) {
	    throw std::invalid_argument("Cannot calculate similarity with a zero vector");
	}

	return dot_product(A, B) / (magA * magB);
}

EmbeddingModel::EmbeddingModel(const std::string& tokenizer_path)
    : env(ORT_LOGGING_LEVEL_WARNING, "HyperSearch")
{
    // ------------------------------------------------------------
    // 1. Load tokenizer JSON
    // ------------------------------------------------------------

    std::ifstream file(tokenizer_path);

    if (!file.is_open()) {
        throw std::runtime_error(
            "Failed to open tokenizer file: " + tokenizer_path
        );
    }

    std::ostringstream output;
    output << file.rdbuf();

    std::string json_blob = output.str();

    tokenizer = tokenizers::Tokenizer::FromBlobJSON(json_blob);


    // ------------------------------------------------------------
    // 2. Load ONNX model
    // ------------------------------------------------------------

    Ort::SessionOptions session_options;

    session = std::make_unique<Ort::Session>(
        env,
        "third_party/models/all-MiniLM-L6-v2/onnx/model.onnx",
        session_options
    );
}


Embedding EmbeddingModel::embed(const std::string& text)
{
    // ------------------------------------------------------------
    // 3. Convert text → token IDs
    // ------------------------------------------------------------

    std::vector<int32_t> token_ids = tokenizer->Encode(text);

    // The tokenizer gives us int32_t IDs.
    // The ONNX model expects int64_t IDs.
    //
    // BEFORE:
    // token_ids = vector<int32_t>
    //
    // AFTER:
    // input_ids = vector<int64_t>

    std::vector<int64_t> input_ids(
        token_ids.begin(),
        token_ids.end()
    );


    // ------------------------------------------------------------
    // 4. Create attention mask
    // ------------------------------------------------------------

    // Our tokenizer uses:
    // pad_id = 0
    //
    // Therefore:
    // real token → 1
    // padding    → 0

    std::vector<int64_t> attention_mask;

    for (int64_t id : input_ids) {
        attention_mask.push_back(id == 0 ? 0 : 1);
    }


    // ------------------------------------------------------------
    // 5. Create token type IDs
    // ------------------------------------------------------------

    // This model receives one sentence.
    // Therefore every token belongs to segment 0.

    std::vector<int64_t> token_type_ids(
        input_ids.size(),
        0
    );


    // ------------------------------------------------------------
    // 6. Describe the tensor shape
    // ------------------------------------------------------------

    // We send:
    //
    // 1 sentence
    // ×
    // sequence_length token positions

    std::array<int64_t, 2> input_shape = {
        1,
        static_cast<int64_t>(input_ids.size())
    };


    // ------------------------------------------------------------
    // 7. Describe where the tensor data lives
    // ------------------------------------------------------------

    // input_ids, attention_mask, and token_type_ids
    // are ordinary CPU vectors.
    //
    // Tell ONNX Runtime that the tensor memory is CPU memory.

    Ort::MemoryInfo memory_info =
        Ort::MemoryInfo::CreateCpu(
            OrtArenaAllocator,
            OrtMemTypeDefault
        );


    // ------------------------------------------------------------
    // 8. Wrap our vectors in ONNX Runtime tensors
    // ------------------------------------------------------------

    Ort::Value input_ids_tensor =
        Ort::Value::CreateTensor<int64_t>(
            memory_info,
            input_ids.data(),
            input_ids.size(),
            input_shape.data(),
            input_shape.size()
        );

    Ort::Value attention_mask_tensor =
        Ort::Value::CreateTensor<int64_t>(
            memory_info,
            attention_mask.data(),
            attention_mask.size(),
            input_shape.data(),
            input_shape.size()
        );

    Ort::Value token_type_ids_tensor =
        Ort::Value::CreateTensor<int64_t>(
            memory_info,
            token_type_ids.data(),
            token_type_ids.size(),
            input_shape.data(),
            input_shape.size()
        );


    // ------------------------------------------------------------
    // 9. Give ONNX Runtime the model's input names
    // ------------------------------------------------------------

    const char* input_names[] = {
        "input_ids",
        "attention_mask",
        "token_type_ids"
    };


    // ------------------------------------------------------------
    // 10. Put the tensors into an array
    // ------------------------------------------------------------

    Ort::Value input_tensors[] = {
        std::move(input_ids_tensor),
        std::move(attention_mask_tensor),
        std::move(token_type_ids_tensor)
    };


    // ------------------------------------------------------------
    // 11. Tell the model which output we want
    // ------------------------------------------------------------

    const char* output_names[] = {
        "last_hidden_state"
    };


    // ------------------------------------------------------------
    // 12. Run the Transformer
    // ------------------------------------------------------------

    auto output_tensors = session->Run(
        Ort::RunOptions{nullptr},
        input_names,
        input_tensors,
        3,
        output_names,
        1
    );


    // ------------------------------------------------------------
    // 13. Get the model's output tensor
    // ------------------------------------------------------------

    const auto& output_tensor = output_tensors[0];


    // The model output is:
    //
    // [batch, sequence_length, hidden_size]
    //
    // [1, 128, 384]

    auto output_shape =
        output_tensor
            .GetTensorTypeAndShapeInfo()
            .GetShape();


    // ------------------------------------------------------------
    // 14. Get a pointer to the output numbers
    // ------------------------------------------------------------

    const float* output_data =
        output_tensor.GetTensorData<float>();


    // ------------------------------------------------------------
    // 15. Mean-pool the token embeddings
    // ------------------------------------------------------------

    constexpr size_t embedding_size = 384;

    std::vector<float> embedding(
        embedding_size,
        0.0f
    );

    size_t token_count = 0;

    for (size_t token = 0;
         token < input_ids.size();
         ++token)
    {
        // Ignore padding positions.

        if (attention_mask[token] == 0)
            continue;


        // Add this token's 384 numbers
        // to our running total.

        for (size_t dim = 0;
             dim < embedding_size;
             ++dim)
        {
            embedding[dim] +=
                output_data[
                    token * embedding_size + dim
                ];
        }

        ++token_count;
    }


    // Divide every dimension by the number
    // of real tokens.
    //
    // This gives us the average token embedding.

    for (float& value : embedding) {
        value /= static_cast<float>(token_count);
    }


    // ------------------------------------------------------------
    // 16. L2-normalize the embedding
    // ------------------------------------------------------------

    float norm = 0.0f;

    for (float value : embedding) {
        norm += value * value;
    }

    norm = std::sqrt(norm);


    for (float& value : embedding) {
        value /= norm;
    }


    // ------------------------------------------------------------
    // 17. Return the final embedding
    // ------------------------------------------------------------

    return Embedding(std::move(embedding));
}

std::vector<Embedding> EmbeddingModel::embed_batch(
    const std::vector<std::string>& texts)
{
    if (texts.empty()) {
        return {};
    }

    const std::size_t batch_size = texts.size();

    // Tokenize every document.
    std::vector<std::vector<int64_t>> tokenized(batch_size);

    std::size_t max_length = 0;

    for (std::size_t i = 0; i < batch_size; ++i) {

        auto token_ids = tokenizer->Encode(texts[i]);

        tokenized[i] =
            std::vector<int64_t>(
                token_ids.begin(),
                token_ids.end()
            );

        max_length =
            std::max(max_length, tokenized[i].size());
    }

    // Flatten the batch into one tensor.
    std::vector<int64_t> input_ids(
        batch_size * max_length,
        0
    );

    std::vector<int64_t> attention_mask(
        batch_size * max_length,
        0
    );

    std::vector<int64_t> token_type_ids(
        batch_size * max_length,
        0
    );

    // Copy each document into its padded row.
    for (std::size_t i = 0; i < batch_size; ++i) {

        for (std::size_t j = 0;
             j < tokenized[i].size();
             ++j)
        {
            input_ids[i * max_length + j] =
                tokenized[i][j];

            attention_mask[i * max_length + j] = 1;
        }
    }

    std::array<int64_t, 2> input_shape = {
        static_cast<int64_t>(batch_size),
        static_cast<int64_t>(max_length)
    };

    Ort::MemoryInfo memory_info =
        Ort::MemoryInfo::CreateCpu(
            OrtArenaAllocator,
            OrtMemTypeDefault
        );

    auto input_ids_tensor =
        Ort::Value::CreateTensor<int64_t>(
            memory_info,
            input_ids.data(),
            input_ids.size(),
            input_shape.data(),
            input_shape.size()
        );

    auto attention_mask_tensor =
        Ort::Value::CreateTensor<int64_t>(
            memory_info,
            attention_mask.data(),
            attention_mask.size(),
            input_shape.data(),
            input_shape.size()
        );

    auto token_type_ids_tensor =
        Ort::Value::CreateTensor<int64_t>(
            memory_info,
            token_type_ids.data(),
            token_type_ids.size(),
            input_shape.data(),
            input_shape.size()
        );

    const char* input_names[] = {
        "input_ids",
        "attention_mask",
        "token_type_ids"
    };

    Ort::Value input_tensors[] = {
        std::move(input_ids_tensor),
        std::move(attention_mask_tensor),
        std::move(token_type_ids_tensor)
    };

    const char* output_names[] = {
        "last_hidden_state"
    };

    auto output_tensors = session->Run(
        Ort::RunOptions{nullptr},
        input_names,
        input_tensors,
        3,
        output_names,
        1
    );

    const auto& output_tensor =
        output_tensors[0];

    const float* output_data =
        output_tensor.GetTensorData<float>();

    constexpr std::size_t embedding_size = 384;

    std::vector<Embedding> results;
    results.reserve(batch_size);

    // Pool each document independently.
    for (std::size_t i = 0; i < batch_size; ++i) {

        std::vector<float> embedding(
            embedding_size,
            0.0f
        );

        std::size_t token_count = 0;

        for (std::size_t token = 0;
             token < max_length;
             ++token)
        {
            if (attention_mask[
                    i * max_length + token] == 0)
            {
                continue;
            }

            const std::size_t offset =
                i * max_length * embedding_size
                + token * embedding_size;

            for (std::size_t dim = 0;
                 dim < embedding_size;
                 ++dim)
            {
                embedding[dim] +=
                    output_data[offset + dim];
            }

            ++token_count;
        }

        if (token_count == 0) {
            throw std::runtime_error(
                "Document produced zero tokens"
            );
        }

        // Mean pooling.
        for (float& value : embedding) {
            value /= static_cast<float>(token_count);
        }

        // L2 normalization.
        float norm = 0.0f;

        for (float value : embedding) {
            norm += value * value;
        }

        norm = std::sqrt(norm);

        for (float& value : embedding) {
            value /= norm;
        }

        results.emplace_back(
            std::move(embedding)
        );
    }

    return results;
}

#include "vector_store.hpp"
#include "similarity.hpp"
#include <algorithm>
#include <omp.h>
#include <omp.h>

#include <limits>
#include <stdexcept>


// Store text together with its embedding.
void VectorStore::add(
    const std::string& text,
    const Embedding& embedding)
{
    documents.push_back({text, embedding});
}


// Return number of stored documents.
std::size_t VectorStore::size() const
{
    return documents.size();
}


// Find the document with the highest similarity.
SearchResult VectorStore::search(const Embedding& Q)
{
    if (documents.empty()) {
        throw std::runtime_error(
            "Vector store is empty"
        );
    }

    // Each OpenMP thread gets its own best score/index.
    // This avoids multiple threads modifying the same
    // variables while they search.
    const int thread_count =
        omp_get_max_threads();

    std::vector<float> best_scores(
        thread_count,
        std::numeric_limits<float>::lowest()
    );

    std::vector<std::size_t> best_indices(
        thread_count,
        0
    );

    // Split the document loop across CPU threads.
    #pragma omp parallel
    {
        const int thread_id =
            omp_get_thread_num();

        #pragma omp for
        for (std::size_t i = 0;
             i < documents.size();
             ++i)
        {
            float score =
                cosine_similarity(
                    documents[i].embedding,
                    Q
                );

            // Each thread only modifies its own slot.
            if (score > best_scores[thread_id]) {
                best_scores[thread_id] = score;
                best_indices[thread_id] = i;
            }
        }
    }

    // Combine each thread's local winner.
    float highest_similarity =
        std::numeric_limits<float>::lowest();

    std::size_t best_index = 0;

    for (int thread = 0;
         thread < thread_count;
         ++thread)
    {
        if (best_scores[thread] > highest_similarity) {
            highest_similarity =
                best_scores[thread];

            best_index =
                best_indices[thread];
        }
    }

    return {
        documents[best_index].text,
        highest_similarity
    };
}

std::vector<SearchResult> VectorStore::search_top_k(
    const Embedding& Q,
    std::size_t k)
{
    if (documents.empty()) {
        throw std::runtime_error(
            "Vector store is empty"
        );
    }

    std::vector<SearchResult> results;

    // Calculate similarity for every document.
    for (const auto& document : documents) {
        float score =
            cosine_similarity(document.embedding, Q);

        results.push_back({
            document.text,
            score
        });
    }

    // Highest similarity first.
    std::sort(
        results.begin(),
        results.end(),
        [](const SearchResult& a,
           const SearchResult& b)
        {
            return a.score > b.score;
        }
    );

    // Don't request more results than exist.
    if (k > results.size()) {
        k = results.size();
    }

    results.resize(k);

    return results;
}

#include "vector_store.hpp"
#include "embedding.hpp"

#include <chrono>
#include <iomanip>
#include <iostream>
#include <random>
#include <string>
#include <vector>
#include <algorithm>

using Clock = std::chrono::steady_clock;


// Generate deterministic synthetic embeddings.
//
// We use 384 dimensions because all-MiniLM-L6-v2
// produces 384-dimensional vectors.
//
// This benchmark is intentionally NOT measuring model
// inference. It measures the performance of the
// VectorStore search itself.
std::vector<Embedding> generate_embeddings(std::size_t count)
{
    constexpr std::size_t dimensions = 384;

    std::mt19937 generator(42);
    std::uniform_real_distribution<float> distribution(-1.0f, 1.0f);

    std::vector<Embedding> embeddings;
    embeddings.reserve(count);

    for (std::size_t i = 0; i < count; ++i) {

        std::vector<float> values;
        values.reserve(dimensions);

        for (std::size_t d = 0; d < dimensions; ++d) {
            values.push_back(distribution(generator));
        }

        embeddings.emplace_back(
            std::move(values)
        );
    }

    return embeddings;
}


// Benchmark VectorStore search at a specific index size.
void benchmark_search(
    const std::vector<Embedding>& embeddings,
    std::size_t count)
{
    VectorStore store;

    // Build an index containing exactly `count` vectors.
    for (std::size_t i = 0; i < count; ++i) {
        store.add(
            std::to_string(i),
            embeddings[i]
        );
    }

    // Use one existing vector as the query.
    const Embedding& query = embeddings[0];

    // Warm up CPU caches and initialization paths.
    store.search(query);

    constexpr std::size_t iterations = 100;

    std::vector<double> times;
    times.reserve(iterations);

    for (std::size_t i = 0; i < iterations; ++i) {

        auto start = Clock::now();

        auto result = store.search(query);

        auto finish = Clock::now();

        // Prevent the compiler from optimizing away the search.
        volatile float score = result.score;
        (void)score;

        double microseconds =
            std::chrono::duration<double, std::micro>(
                finish - start
            ).count();

        times.push_back(microseconds);
    }

    std::sort(
        times.begin(),
        times.end()
    );

    double p50 =
        times[times.size() / 2];

    double p95 =
        times[
            static_cast<std::size_t>(
                times.size() * 0.95
            )
        ];

    double p99 =
        times[
            static_cast<std::size_t>(
                times.size() * 0.99
            )
        ];

    double qps =
        1'000'000.0 / p50;

    std::cout
        << std::left
        << std::setw(12)
        << count
        << std::setw(14)
        << std::fixed
        << std::setprecision(2)
        << p50
        << std::setw(14)
        << p95
        << std::setw(14)
        << p99
        << qps
        << '\n';
}


int main()
{
    constexpr std::size_t maximum_vectors = 100000;

    std::cout
        << "========================================\n"
        << "HyperSearch Search Benchmark\n"
        << "========================================\n\n";

    std::cout
        << "Generating "
        << maximum_vectors
        << " synthetic 384-dimensional embeddings...\n";

    auto embeddings =
        generate_embeddings(maximum_vectors);

    std::cout
        << "Generated "
        << embeddings.size()
        << " vectors\n\n";


    std::cout
        << "Search scaling\n"
        << "--------------------------------------------------\n"
        << std::left
        << std::setw(12) << "Vectors"
        << std::setw(14) << "p50 us"
        << std::setw(14) << "p95 us"
        << std::setw(14) << "p99 us"
        << "QPS\n";

    std::cout
        << "--------------------------------------------------\n";


    const std::vector<std::size_t> sizes = {
        1000,
        10000,
        100000
    };

    for (std::size_t count : sizes) {

        benchmark_search(
            embeddings,
            count
        );
    }


    std::cout
        << "\nBenchmark complete.\n";
}

#include "embedding_model.hpp"
#include "vector_store.hpp"

#include <fstream>
#include <iostream>
#include <sstream>
#include <stdexcept>
#include <string>
#include <vector>
#include <algorithm>


// Document used by the test.
struct TestDocument {
    std::string id;
    std::string text;
};


// Query and its expected document.
struct TestQuery {
    std::string id;
    std::string text;
    std::string expected_id;
};


// Load the corpus.
std::vector<TestDocument> load_corpus(
    const std::string& path)
{
    std::ifstream file(path);

    if (!file)
        throw std::runtime_error("Could not open corpus");

    std::vector<TestDocument> documents;

    std::string line;
    std::string id;
    std::string text;

    while (std::getline(file, line)) {

        // Start of a new document.
        if (line.rfind("[DOC_", 0) == 0) {

            // Save the previous document.
            if (!id.empty()) {
                documents.push_back({id, text});
                text.clear();
            }

            // Remove '[' and ']'.
            id = line.substr(1, line.size() - 2);

            continue;
        }

        // Add document text.
        if (!line.empty())
            text += line;
    }

    // Save the last document.
    if (!id.empty())
        documents.push_back({id, text});

    return documents;
}


// Load evaluation queries.
std::vector<TestQuery> load_queries(
    const std::string& path)
{
    std::ifstream file(path);

    if (!file)
        throw std::runtime_error("Could not open queries");

    std::vector<TestQuery> queries;
    std::string line;

    while (std::getline(file, line)) {

        if (line.empty())
            continue;

        std::stringstream stream(line);

        TestQuery query;

        // Format: ID|query|expected document.
        std::getline(stream, query.id, '|');
        std::getline(stream, query.text, '|');
        std::getline(stream, query.expected_id, '|');

        queries.push_back(query);
    }

    return queries;
}


int main()
{
    // Load the model.
    EmbeddingModel model(
        "third_party/models/all-MiniLM-L6-v2/tokenizer.json"
    );

    VectorStore store;

    // Load test documents.
    auto documents =
        load_corpus("tests/corpus.txt");

    std::cout
        << "Documents: "
        << documents.size()
        << '\n';

    // Map document text back to its test ID.
    std::vector<std::pair<std::string, std::string>> id_lookup;

    // Embed every document.
    for (const auto& document : documents) {

        Embedding embedding =
            model.embed(document.text);

        store.add(
            document.text,
            embedding
        );

        id_lookup.push_back({
            document.text,
            document.id
        });
    }

    // Load queries.
    auto queries =
        load_queries("tests/queries.txt");

    std::cout
        << "Queries: "
        << queries.size()
        << "\n\n";


    int recall_at_1 = 0;
    int recall_at_5 = 0;

    double reciprocal_rank_sum = 0.0;


    // Run every evaluation query.
    for (const auto& query : queries) {

        Embedding query_embedding =
            model.embed(query.text);

        auto results =
            store.search_top_k(
                query_embedding,
                5
            );

        int rank = 0;

        // Find where the expected document appeared.
        for (const auto& result : results) {

            ++rank;

            auto it = std::find_if(
                id_lookup.begin(),
                id_lookup.end(),
                [&](const auto& entry)
                {
                    return entry.first == result.text;
                }
            );

            if (it != id_lookup.end() &&
                it->second == query.expected_id) {

                if (rank == 1)
                    ++recall_at_1;

                ++recall_at_5;

                reciprocal_rank_sum +=
                    1.0 / rank;

                break;
            }
        }

        std::cout
            << query.id
            << " -> ";

        if (rank == 0)
            std::cout << "MISS";
        else
            std::cout << "rank " << rank;

        std::cout << '\n';
    }


    // Calculate retrieval metrics.
    double total =
        static_cast<double>(queries.size());

    double recall1 =
        recall_at_1 / total;

    double recall5 =
        recall_at_5 / total;

    double mrr =
        reciprocal_rank_sum / total;


    std::cout << "\n";
    std::cout << "============================\n";
    std::cout << "HyperSearch Retrieval Test\n";
    std::cout << "============================\n";

    std::cout
        << "Recall@1: "
        << recall1 * 100
        << "%\n";

    std::cout
        << "Recall@5: "
        << recall5 * 100
        << "%\n";

    std::cout
        << "MRR: "
        << mrr
        << '\n';
}
